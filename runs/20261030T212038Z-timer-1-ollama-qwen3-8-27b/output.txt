=== tsccal.c ===
/*
 * tsccal.c - TSC rate measured against the 8254 PIT.
 *
 * Lifted from Linux arch/x86/kernel/tsc.c:
 *   pit_calibrate_tsc() and pit_hpet_ptimer_calibrate_cpu()
 *
 * Measure how many CPU cycles pass in a millisecond by timing against
 * the PC's 8254 timer. If reading the 8254 is too slow/unreliable
 * (a virtualized machine), give up on that and take the hypervisor's
 * word on the TSC rate (CPUID).
 *
 * The x86 port instructions and the TSC read are written as inline
 * assembly; there are no kernel headers available.
 */
#include "tsccal.h"

#define PIT_TICK_RATE  1193182UL
#define CAL_MS         10
#define CAL_LATCH      (PIT_TICK_RATE / (1000 / CAL_MS))
#define CAL_PIT_LOOPS  1000
#define CAL2_MS        50
#define CAL2_LATCH     (PIT_TICK_RATE / (1000 / CAL2_MS))
#define CAL2_PIT_LOOPS 5000

/* ---- raw x86 primitives, inline assembly ---- */

static unsigned char tsc_inb(unsigned int port)
{
	unsigned char v;
	__asm__ __volatile__ ("inb %w1, %b0" : "=a" (v) : "d" (port));
	return v;
}

static void tsc_outb(unsigned char v, unsigned int port)
{
	__asm__ __volatile__ ("outb %b0, %w1" : : "a" (v), "d" (port));
}

static unsigned long long tsc_rdtsc(void)
{
	unsigned int lo, hi;
	__asm__ __volatile__ ("rdtsc" : "=a" (lo), "=d" (hi));
	return ((unsigned long long)hi << 32) | (unsigned long long)lo;
}

/* ---- CPUID fallback: the hypervisor's word on the TSC rate ---- */

static unsigned long cpuid_tsc_khz(void)
{
	unsigned int max_leaf, eax, ebx, ecx, edx;

	__asm__ __volatile__ ("cpuid"
			     : "=a" (max_leaf), "=b" (ebx), "=c" (ecx), "=d" (edx)
			     : "a" (0));

	/* CPUID 0x15: TSC/crystal ratio (ebx/eax) and crystal Hz (ecx). */
	if (max_leaf >= 0x15) {
		unsigned int denom, numer, crystal_hz;
		__asm__ __volatile__ ("cpuid"
				: "=a" (denom), "=b" (numer), "=c" (crystal_hz), "=d" (edx)
				: "a" (0x15));
		if (numer && denom && crystal_hz) {
			unsigned long khz = (unsigned long)(crystal_hz / 1000UL);
			khz = khz * numer / denom;
			if (khz)
				return khz;
		}
	}

	/* CPUID 0x16: base frequency in MHz (eax). */
	if (max_leaf >= 0x16) {
		unsigned int base_mhz;
		__asm__ __volatile__ ("cpuid"
				: "=a" (base_mhz), "=b" (ebx), "=c" (ecx), "=d" (edx)
				: "a" (0x16));
		if (base_mhz)
			return base_mhz * 1000UL;
	}

	return 0;
}

/* ---- PIT calibration ---- */

static unsigned long pit_calibrate(unsigned long latch,
				   unsigned long ms,
				   unsigned int loopmin)
{
	unsigned long long tsc, t1, t2, delta;
	unsigned long tscmin, tscmax;
	unsigned int pitcnt;

	/* Set the Gate high, disable speaker */
	tsc_outb((unsigned char)((tsc_inb(0x61) & 0xFD) | 0x01), 0x61);

	/* Setup CTC channel 2 for mode 0 (terminal count), binary. LSB then MSB. */
	tsc_outb(0xb0, 0x43);
	tsc_outb((unsigned char)(latch & 0xFF), 0x42);
	tsc_outb((unsigned char)(latch >> 8), 0x42);

	tsc = t1 = t2 = tsc_rdtsc();

	pitcnt = 0;
	tscmax = 0;
	tscmin = (unsigned long)-1;
	while ((tsc_inb(0x61) & 0x20) == 0) {
		t2 = tsc_rdtsc();
		delta = t2 - tsc;
		tsc = t2;
		if ((unsigned long)delta < tscmin)
			tscmin = (unsigned long)delta;
		if ((unsigned long)delta > tscmax)
			tscmax = (unsigned long)delta;
		pitcnt++;
	}

	/*
	 * Sanity checks:
	 * If we were not able to read the PIT more than loopmin times,
	 * then we have been hit by a massive SMI.
	 * If the maximum is 10 times larger than the minimum, then we
	 * got hit by an SMI as well.
	 */
	if (pitcnt < loopmin || tscmax > 10 * tscmin)
		return 0;

	delta = t2 - t1;
	return (unsigned long)(delta / ms);
}

/* ---- public interface ---- */

unsigned long tsccal_khz(void)
{
	unsigned long tsc_pit_min = (unsigned long)-1;
	unsigned long latch = CAL_LATCH;
	unsigned long ms = CAL_MS;
	unsigned int loopmin = CAL_PIT_LOOPS;
	int i;

	/* Run 3 loops, keeping the lowest (best) PIT estimate. */
	for (i = 0; i < 3; i++) {
		unsigned long tsc_pit_khz = pit_calibrate(latch, ms, loopmin);

		/* Pick the lowest PIT TSC calibration so far */
		if (tsc_pit_khz != 0 && tsc_pit_khz < tsc_pit_min)
			tsc_pit_min = tsc_pit_khz;

		/* PIT keeps failing: give the virtual PC a longer window. */
		if (i == 1 && tsc_pit_min == (unsigned long)-1) {
			latch = CAL2_LATCH;
			ms = CAL2_MS;
			loopmin = CAL2_PIT_LOOPS;
		}
	}

	/* Measured against the PIT: use it. */
	if (tsc_pit_min != (unsigned long)-1)
		return tsc_pit_min;

	/*
	 * Reading the 8254 was too slow/unreliable (virtualized
	 * environment). Give up on measuring and take the hypervisor's
	 * word on the TSC frequency.
	 */
	return cpuid_tsc_khz();
}

unsigned long long tsccal_cycles(void)
{
	return tsc_rdtsc();
}
