=== tsccal.c ===
/*
 * tsccal.c - the processor counter's (TSC) rate, measured against the 8254.
 *
 * Lifted from the Linux kernel's TSC calibration (arch/x86/kernel/tsc.c):
 * the "fast" path times TSC reads against the 8254 (channel 2 counter,
 * port 0x42), and when the reads are too slow (a virtualised PIT) it gives
 * up and takes the hypervisor's word on the TSC frequency from CPUID.
 *
 * Freestanding 32-bit x86: no kernel headers, port I/O via inline assembly.
 */

#include "tsccal.h"

#define TS_RATE 1193182UL
#define TS_MAX_MS 50
#define TS_MAX_IT (TS_MAX_MS * TS_RATE / 1000 / 256)

/* ---- primitives --------------------------------------------------------- */

static unsigned char t_inb(unsigned short port)
{
	unsigned char ret;
	__asm__ __volatile__ ("inb %1, %0" : "=a"(ret) : "Nd"(port) : "memory");
	return ret;
}

static void t_outb(unsigned char val, unsigned short port)
{
	__asm__ __volatile__ ("outb %0, %1" : : "a"(val), "Nd"(port) : "memory");
}

static unsigned long long t_rdtsc(void)
{
	unsigned int lo, hi;
	__asm__ __volatile__ ("rdtsc" : "=a"(lo), "=d"(hi));
	return ((unsigned long long)hi << 32) | (unsigned long long)lo;
}

static void t_cpuid(unsigned int leaf, unsigned int *a, unsigned int *b,
		    unsigned int *c, unsigned int *d)
{
	__asm__ __volatile__ ("cpuid"
			      : "=a"(*a), "=b"(*b), "=c"(*c), "=d"(*d)
			      : "a"(leaf)
			      : "memory");
}

/* ---- fast path: TSC against the 8254 ------------------------------------ */

/* Read channel-2 (0x42) LSB then MSB; return whether MSB == v. */
static int t_pit_msb(unsigned char v)
{
	unsigned char hi;
	(void)t_inb(0x42);	/* drop LSB */
	hi = t_inb(0x42);	/* MSB */
	return hi == v;
}

/*
 * Poll until the door's MSB changes away from v, sampling TSC between
 * reads.  *tp = TSC at transition; *dp = last fast sample distance.
 * Return nonzero if enough samples to trust the timing.
 */
static int t_pit_wait(unsigned char v, unsigned long long *tp, unsigned long *dp)
{
	int n;
	unsigned long long t = 0, pt = 0;

	for (n = 0; n < 50000; n++) {
		if (!t_pit_msb(v))
			break;
		pt = t;
		t = t_rdtsc();
	}
	*dp = (unsigned long)(t_rdtsc() - pt);
	*tp = t;
	return n > 5;
}

/*
 * Fast TSC calibration (kernel quick_pit_calibrate): latch channel 2 to
 * 0xffff, mode 0, watch MSB fall, use TSC at transitions.
 * Returns kHz or 0 if the door is too slow to measure.
 */
static unsigned long t_fast(void)
{
	int i, ok = 0;
	unsigned long long t, delta;
	unsigned long d1, d2;

	/* Gate high, speaker off (port 0x61). */
	t_outb((unsigned char)((t_inb(0x61) & 0xfd) | 0x01), 0x61);

	/* Channel 2, mode 0, binary (port 0x43). */
	t_outb(0xb0, 0x43);

	/* Latch 0xffff: LSB then MSB (port 0x42). */
	t_outb(0xff, 0x42);
	t_outb(0xff, 0x42);

	/* Wait for first edge. */
	t_pit_msb(0);

	if (t_pit_wait(0xff, &t, &d1)) {
		for (i = 1; i <= TS_MAX_IT; i++) {
			if (!t_pit_wait((unsigned char)(0xff - i), &delta, &d2))
				break;

			delta -= t;

			if (i == 1 && d1 + d2 >= (delta * TS_MAX_IT) >> 11)
				return 0;

			if (d1 + d2 >= delta >> 11)
				continue;

			if (!t_pit_msb((unsigned char)(0xfe - i)))
				break;
			ok = 1;
			break;
		}
	}

	if (!ok)
		return 0;

	delta *= TS_RATE;
	delta /= (unsigned long long)i * 256 * 1000ULL;
	return (unsigned long)delta;
}

/* ---- fallback: hypervisor's word via CPUID ------------------------------- */

static unsigned long t_cpu_khz(void)
{
	unsigned int de, nu, ck, un, bm;

	t_cpuid(0x15, &de, &nu, &ck, &un);
	if (nu == 0 || de == 0)
		return 0;

	ck /= 1000;	/* Hz -> kHz */

	if (ck == 0) {
		t_cpuid(0x16, &bm, &un, &un, &un);
		if (bm == 0)
			return 0;
		ck = (unsigned int)((unsigned long long)bm * 1000ULL * de / nu);
	}

	if (ck == 0)
		return 0;

	return (unsigned long)((unsigned long long)ck * nu / de);
}

/* ---- public interface ---------------------------------------------------- */

unsigned long tsccal_khz(void)
{
	unsigned long k = t_fast();
	if (k)
		return k;
	return t_cpu_khz();
}

unsigned long long tsccal_cycles(void)
{
	return t_rdtsc();
}
