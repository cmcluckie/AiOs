=== tsccal.c ===
/*
 * tsccal.c — measure the TSC rate against the 8254 PIT.
 *
 * Lifted from arch/x86/kernel/tsc.c (pit_calibrate_tsc).  We latch
 * PIT channel 2 for 10 ms (mode 0, terminal count), then poll the
 * 8254 state register (port 0x61, bit 5 — channel 2 output) for the
 * terminal-count edge.  Each loop iteration captures the TSC *before*
 * any further I/O, and the min/max of consecutive deltas are tracked
 * so that a slow or disturbed PIT read (the situation on a
 * hypervisor) is detected and the measurement is abandoned, returning
 * 0 so the caller takes the hypervisor's word.
 *
 * No kernel headers, no C library.  32-bit freestanding x86.
 */
#include "tsccal.h"

/* ---- inline port I/O and TSC readout ---- */

static inline unsigned char inb(unsigned short port)
{
	unsigned char v;
	__asm__ __volatile__ ("inb %w1, %0"
			       : "=a" (v)
			       : "Nd" (port));
	return v;
}

static inline void outb(unsigned char v, unsigned short port)
{
	__asm__ __volatile__ ("outb %0, %w1"
			       :
			       : "a" (v), "Nd" (port));
}

static inline unsigned long long rdtsc(void)
{
	unsigned int lo, hi;
	__asm__ __volatile__ ("rdtsc"
			       : "=a" (lo), "=d" (hi));
	return ((unsigned long long)hi << 32) | lo;
}

/* ---- calibration ---- */

#define PIT_TICK_RATE	1193182	/* Hz */
#define CAL_MS		10		/* latch window, ms */
#define CAL_LATCH	(PIT_TICK_RATE / (1000 / CAL_MS))
#define CAL_PIT_LOOPS	1000	/* minimum PIT reads we must observe */

unsigned long tsccal_khz(void)
{
	unsigned long long tsc, t1, t2, delta;
	unsigned long tscmin, tscmax;
	int pitcnt;

	/* Set the gate high (0x61 bit 0), disable speaker (bit 1 clear). */
	outb((unsigned char)((inb(0x61) & ~0x02) | 0x01), 0x61);

	/*
	 * CTC channel 2, mode 0 (interrupt on terminal count), binary.
	 * Latch = CAL_MS worth of PIT ticks (LSB then MSB).
	 */
	outb(0xb0, 0x43);
	outb((unsigned char)(CAL_LATCH & 0xff), 0x42);
	outb((unsigned char)(CAL_LATCH >> 8), 0x42);

	tsc = t1 = t2 = rdtsc();

	pitcnt = 0;
	tscmax = 0;
	tscmin = 0xffffffffUL;
	while ((inb(0x61) & 0x20) == 0) {
		/* Capture TSC *before* the next I/O. */
		t2 = rdtsc();
		delta = t2 - tsc;
		tsc = t2;
		if ((unsigned long)delta < tscmin)
			tscmin = (unsigned long)delta;
		if ((unsigned long)delta > tscmax)
			tscmax = (unsigned long)delta;
		pitcnt++;
	}

	/*
	 * Sanity:
	 *  - we must have read the PIT at least CAL_PIT_LOOPS times
	 *    (fewer means the reads were too slow, i.e. we are on a
	 *    hypervisor where the 8254 is emulated and slow);
	 *  - the max per-iteration delta must be within a factor of
	 *    10 of the min (a large spread means an SMI or other
	 *    disturbance hit the measurement).
	 *
	 * In either case the measurement is abandoned and 0 is
	 * returned, so the caller takes the hypervisor's word.
	 */
	if (pitcnt < CAL_PIT_LOOPS || tscmax > 10 * tscmin)
		return 0;

	/* TSC ticks per millisecond over the whole CAL_MS window. */
	delta = t2 - t1;
	return (unsigned long)(delta / CAL_MS);
}

unsigned long tsccal_cycles(void)
{
	return rdtsc();
}
