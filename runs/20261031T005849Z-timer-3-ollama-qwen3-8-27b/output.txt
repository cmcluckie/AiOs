=== tsccal.c ===
#include "tsccal.h"

/* Constants from include/linux/timex.h and arch/x86/kernel/tsc.c */
#define PIT_TICK_RATE	1193182UL
#define CAL2_MS		50
#define CAL2_LATCH	(PIT_TICK_RATE / (1000 / CAL2_MS))
#define CAL2_PIT_LOOPS	5000

/* --- port I/O (no libc, no OS) --- */

static unsigned char inb(unsigned short port)
{
	unsigned char ret;
	asm volatile ("inb %w1, %0" : "=a" (ret) : "dN" (port));
	return ret;
}

static void outb(unsigned char val, unsigned short port)
{
	asm volatile ("outb %0, %w1" : : "a" (val), "dN" (port) : "memory");
}

/* --- TSC read --- */

static unsigned long long rdtscl(void)
{
	unsigned long long val;
	asm volatile ("rdtsc" : "=A" (val) : : "memory");
	return val;
}

unsigned long long tsccal_cycles(void)
{
	return rdtscl();
}

/*
 * PIT calibration – lifted from arch/x86/kernel/tsc.c: pit_calibrate_tsc().
 * Sets PIT channel 2 to one-shot mode 0, counts TSC while the output is low,
 * then sanity-checks the per-iteration deltas to catch SMI / slow-I/O cases.
 * Returns TSC kHz (cycles per millisecond) or 0 on failure.
 */
static unsigned long pit_calibrate_tsc(unsigned long latch,
				       unsigned long ms,
				       int loopmin)
{
	unsigned long long tsc, t1, t2, delta;
	unsigned long tscmin, tscmax;
	int pitcnt;

	/* Set the Gate high, disable speaker */
	outb((unsigned char)((inb(0x61) & ~0x02) | 0x01), 0x61);

	/*
	 * Setup CTC channel 2 for mode 0 (interrupt on terminal
	 * count mode), binary count.  Set the latch register to
	 * the requested timeout (LSB then MSB) to begin countdown.
	 */
	outb(0xb0, 0x43);
	outb((unsigned char)(latch & 0xff), 0x42);
	outb((unsigned char)(latch >> 8), 0x42);

	tsc = t1 = t2 = rdtscl();

	pitcnt = 0;
	tscmax = 0;
	tscmin = 0xffffffffUL;
	while ((inb(0x61) & 0x20) == 0) {
		t2 = rdtscl();
		delta = t2 - tsc;
		tsc = t2;
		if ((unsigned long)delta < tscmin)
			tscmin = (unsigned long)delta;
		if ((unsigned long)delta > tscmax)
			tscmax = (unsigned long)delta;
		pitcnt++;
	}

	/*
	 * Sanity checks:
	 *
	 * If we were not able to read the PIT more than loopmin
	 * times, then we have been hit by a massive SMI (or the
	 * hypervisor's emulated port I/O is too slow).
	 *
	 * If the maximum is 10 times larger than the minimum,
	 * then we got hit by an SMI as well.
	 */
	if (pitcnt < loopmin || tscmax > 10 * tscmin)
		return 0;

	/* Calculate the PIT value: cycles / ms */
	delta = t2 - t1;
	return (unsigned long)(delta / ms);
}

/*
 * CPUID fallback – lifted from arch/x86/kernel/tsc.c: native_calibrate_tsc().
 * Uses CPUID leaf 0x15 (TSC/Crystal ratio) and optionally leaf 0x16
 * (base frequency) to obtain the TSC rate without touching the PIT.
 * Returns TSC kHz, or 0 if the CPU does not provide the data.
 */

static unsigned int cpu_max_leaf(void)
{
	unsigned int eax;
	asm volatile ("cpuid" : "=a" (eax) : "a" (0));
	return eax;
}

static unsigned int is_intel(void)
{
	unsigned int ebx, ecx, edx;
	asm volatile ("cpuid" : "=b" (ebx), "=c" (ecx), "=d" (edx) : "a" (0));
	/* "GenuineIntel" = "Genu"(ebx) "ineI"(edx) "ntel"(ecx) */
	return ebx == 0x756e6547 && edx == 0x49656e69 && ecx == 0x6c65746e;
}

static void cpuid15(unsigned int *denom, unsigned int *num, unsigned int *hz)
{
	unsigned int eax, ebx, ecx;
	asm volatile ("cpuid" : "=a" (eax), "=b" (ebx), "=c" (ecx) : "a" (0x15));
	*denom = eax;
	*num = ebx;
	*hz = ecx;
}

static unsigned int cpuid16_base_mhz(void)
{
	unsigned int eax;
	asm volatile ("cpuid" : "=a" (eax) : "a" (0x16));
	return eax;
}

/* INTEL_ATOM_GOLDMONT_D = family 6, model 0x5F */
static int is_denyton(void)
{
	unsigned int eax;
	asm volatile ("cpuid" : "=a" (eax) : "a" (0x1));
	{
		unsigned int fam, mod;
		fam = (eax >> 8) & 0xf;
		if (fam == 0xf)
			fam += (eax >> 20) & 0xff;
		mod = ((eax >> 4) & 0xf) | ((eax & 0xf0000000u) >> 16);
		return fam == 6 && mod == 0x5f;
	}
}

static unsigned long tsc_from_cpuid(void)
{
	unsigned int denom, num, hz;
	unsigned int crystal_khz;

	if (!is_intel())
		return 0;

	if (cpu_max_leaf() < 0x15)
		return 0;

	denom = num = hz = 0;
	cpuid15(&denom, &num, &hz);

	if (num == 0 || denom == 0)
		return 0;

	crystal_khz = hz / 1000;

	/*
	 * Denverton SoCs don't report crystal clock, and also don't
	 * support CPUID_LEAF_FREQ for the calculation below, so
	 * hardcode the 25 MHz crystal clock.
	 */
	if (crystal_khz == 0 && is_denyton())
		crystal_khz = 25000;

	if (crystal_khz == 0 && cpu_max_leaf() >= 0x16) {
		unsigned int eax_base_mhz;

		eax_base_mhz = cpuid16_base_mhz();
		crystal_khz = eax_base_mhz * 1000 *
			      denom / num;
	}

	if (crystal_khz == 0)
		return 0;

	return (unsigned long)((unsigned long long)crystal_khz * num / denom);
}

/* Cached result so we only calibrate once */
static unsigned long tsc_khz_cached;
static int tsc_khz_done;

unsigned long tsccal_khz(void)
{
	unsigned long khz;

	if (tsc_khz_done)
		return tsc_khz_cached;

	tsc_khz_done = 1;
	khz = 0;

	/*
	 * Try to measure the TSC against the PIT.  On a virtualised
	 * machine the port I/O is emulated and very slow, so the
	 * sanity checks in pit_calibrate_tsc() will reject the
	 * result; in that case fall through to the CPUID path
	 * (the hypervisor's own reported frequency).
	 */
	khz = pit_calibrate_tsc(CAL2_LATCH, CAL2_MS, CAL2_PIT_LOOPS);

	if (khz == 0)
		khz = tsc_from_cpuid();

	tsc_khz_cached = khz;
	return khz;
}
