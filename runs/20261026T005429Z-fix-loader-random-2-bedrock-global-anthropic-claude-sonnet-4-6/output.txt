>  * Use rdrand to fill buf with n bytes, 4 bytes at a time.  Returns 1 on
>  * success, 0 if rdrand fails 10 times on any word (RDRAND_RETRY_LOOPS, the
>  * way Linux gives up: arch/x86/include/asm/archrandom.h).
>  */
> static int rdrand_fill(unsigned char *buf, unsigned long n)
> {

> }
