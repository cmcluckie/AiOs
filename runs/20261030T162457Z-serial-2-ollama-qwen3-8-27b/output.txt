=== serial8250.c ===

#include "serial8250.h"

/*
 * 16550 register offsets (regshift 0, plain byte registers), from
 * include/uapi/linux/serial_reg.h.
 */
#define UART_RX		0	/* In:  Receive buffer */
#define UART_TX		0	/* Out: Transmit buffer */

#define UART_IER	1	/* Out: Interrupt Enable Register */
#define UART_IER_UUE	0x40	/* UART Unit Enable (XScale) */

#define UART_FCR	2	/* Out: FIFO Control Register */

#define UART_LCR	3	/* Out: Line Control Register */
#define UART_LCR_DLAB	0x80	/* Divisor latch access bit */
#define UART_LCR_WLEN8	0x03	/* Wordlength: 8 bits */

#define UART_MCR	4	/* Out: Modem Control Register */
#define UART_MCR_RTS	0x02	/* RTS complement */
#define UART_MCR_DTR	0x01	/* DTR complement */

#define UART_LSR	5	/* In:  Line Status Register */
#define UART_LSR_TEMT	0x40	/* Transmitter empty */

#define UART_DLL	0	/* Out: Divisor Latch Low */
#define UART_DLM	1	/* Out: Divisor Latch High */

#define UARTCLK 1843200u	/* 115200 * 16 */
#define BAUD 115200u

static unsigned int serial8250_in(unsigned short port)
{
	unsigned int v;

	__asm__ volatile ("inb %w1, %0" : "=a"(v) : "Nd"(port));
	return v;
}

static void serial8250_out(unsigned short port, unsigned int v)
{
	__asm__ volatile ("outb %b0, %w1" : : "a"(v), "Nd"(port) : "memory");
}

static void serial8250_wait_temt(unsigned short base)
{
	unsigned int status;

	for (;;) {
		status = serial8250_in(base + UART_LSR);
		if (status & UART_LSR_TEMT)
			break;
		__asm__ volatile ("pause" ::: "memory");
	}
}

void serial8250_init(unsigned short base)
{
	unsigned int divisor;
	unsigned char c;
	unsigned int ier;

	serial8250_out(base + UART_LCR, UART_LCR_WLEN8);		/* 8n1 */
	ier = serial8250_in(base + UART_IER);
	serial8250_out(base + UART_IER, ier & UART_IER_UUE);	/* no interrupt */
	serial8250_out(base + UART_FCR, 0);	/* no fifo */
	serial8250_out(base + UART_MCR, UART_MCR_DTR | UART_MCR_RTS);

	divisor = (UARTCLK + 8 * BAUD) / (16 * BAUD);
	c = serial8250_in(base + UART_LCR);
	serial8250_out(base + UART_LCR, c | UART_LCR_DLAB);
	serial8250_out(base + UART_DLL, divisor & 0xff);
	serial8250_out(base + UART_DLM, (divisor >> 8) & 0xff);
	serial8250_out(base + UART_LCR, c & ~UART_LCR_DLAB);
}

void serial8250_putc(unsigned short base, unsigned char c)
{
	serial8250_out(base + UART_TX, c);
	serial8250_wait_temt(base);
}

void serial8250_write(unsigned short base, const void *s, unsigned long n)
{
	const unsigned char *p = s;

	while (n--)
		serial8250_putc(base, *p++);
}

void serial8250_flush(unsigned short base)
{
	serial8250_wait_temt(base);
}
